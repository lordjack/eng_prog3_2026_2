INSERT — sintaxe basica

-- Forma 1: especificando colunas (recomendada)
INSERT INTO usuarios (nome, email, senha_hash)
VALUES ('Ana Silva', 'ana@email.com', 'hash_da_senha');

-- Forma 2: todas as colunas na ordem (fragil!)

VALUES (NULL, 'Bruno Souza', 'bruno@email.com', 'hash', 1, NOW(), NOW());
Sempre liste as colunas
A forma 2 quebra se a tabela mudar. Com colunas explicitas, o INSERT sobrevive a ALTER TABLE.

Retorno O MySQL responde: Query OK, 1 row affected

INSERT com FK e erros comuns

-- Correto: o usuario 1 existe
INSERT INTO tarefas (titulo, descricao, prioridade, id_usuario)
VALUES ('Estudar SQL', 'Resolver os exercicios do E03', 'alta', 1);

-- ERRO 1452: usuario 99 nao existe!
INSERT INTO tarefas (titulo, id_usuario) VALUES ('Tarefa orfa', 99);
-- Cannot add or update a child row: a foreign key constraint fails
Erro 1062 — Duplicate entry
Violou UNIQUE ou PRIMARY KEY. Ex: email ja cadastrado.
Erro 1048 — Cannot be null
Esqueceu coluna NOT NULL sem DEFAULT.

UPDATE — alterando registros

-- Atualiza UM registro (com WHERE na PK)
UPDATE tarefas
SET status = 'concluida'
WHERE id_tarefa = 3;

-- Atualiza VARIOS registros de uma vez
UPDATE tarefas
SET prioridade = 'alta'
WHERE id_usuario = 1 AND status = 'pendente';

-- Atualizacao calculada
UPDATE produtos SET preco = preco * 1.10 WHERE id_cat = 2;
NUNCA esqueca o WHERE!
UPDATE tarefas SET status='concluida'; marca TODAS as tarefas como concluidas. Sem WHERE = desastre.

DELETE — removendo registros

-- Remove UM registro
DELETE FROM tarefas WHERE id_tarefa = 5;

-- Remove por condicao
DELETE FROM tarefas WHERE status = 'concluida';

-- DELETE vs TRUNCATE
DELETE FROM tarefas;    -- linha a linha, pode dar ROLLBACK
TRUNCATE TABLE tarefas; -- recria a tabela, mais rapido, sem rollback
Efeito do CASCADE
DELETE FROM usuarios WHERE id_usuario = 1; apaga tambem TODAS as tarefas dele (ON DELETE CASCADE). Sempre confira o impacto antes!

Aula 2

Seguranca: teste antes com SELECT

Padrao de ouro: SELECT -> DML

-- 1. Veja O QUE sera afetado
SELECT * FROM tarefas WHERE status = 'concluida';

-- 2. Confira a quantidade
SELECT COUNT(*) FROM tarefas WHERE status = 'concluida';

-- 3. So entao execute o DELETE com a MESMA condicao
DELETE FROM tarefas WHERE status = 'concluida';
MySQL Workbench — Safe Updates
Por padrao, o Workbench bloqueia UPDATE/DELETE sem WHERE na PK (erro 1175). Isso e seu amigo, nao seu inimigo!

Transacoes: seu botao de "desfazer"

START TRANSACTION;

UPDATE usuarios SET ativo = 0 WHERE id_usuario = 2;
DELETE FROM tarefas WHERE id_usuario = 2;

-- Algo errado? Desfaz TUDO:
ROLLBACK;

-- Tudo certo? Confirma TUDO:
-- COMMIT;
COMMIT
Confirma as alteracoes definitivamente.
ROLLBACK
Desfaz tudo desde o START TRANSACTION.

Lembrete DDL (CREATE/ALTER/DROP) ignora transacao — so DML pode ser desfeito.

SELECT — anatomia da consulta

SELECT coluna1, coluna2        -- O QUE voce quer ver
FROM tabela                    -- DE ONDE vem
WHERE condicao                 -- FILTRO de linhas
ORDER BY coluna DESC           -- ORDENACAO
LIMIT 10;                      -- QUANTAS linhas

Exemplos basicos

SELECT * FROM usuarios;                       -- tudo
SELECT nome, email FROM usuarios;             -- colunas
SELECT nome FROM usuarios WHERE ativo = 1;    -- filtro
SELECT titulo FROM tarefas ORDER BY criado_em DESC LIMIT 5;

Dica Evite SELECT * em producao — liste so o que precisa.

WHERE — operadores de filtro

Comparacao e logica

WHERE status = 'pendente'
WHERE prioridade <> 'baixa'
WHERE id_usuario = 1 AND status = 'pendente'
WHERE status = 'pendente' OR status = 'em_andamento'

Intervalos e listas

WHERE id_tarefa BETWEEN 5 AND 10
WHERE prioridade IN ('alta','media')
WHERE id_usuario NOT IN (1, 2)

Texto e nulos

WHERE titulo LIKE '%SQL%'     -- contem
WHERE email LIKE 'ana%'       -- comeca
WHERE descricao IS NULL       -- e nulo
WHERE descricao IS NOT NULL   -- tem valor
NULL != vazio
NULL se compara com IS, nunca com = . = NULL sempre retorna falso!

ORDER BY, DISTINCT e LIMIT

Ordenacao

SELECT titulo, prioridade FROM tarefas
ORDER BY prioridade ASC, criado_em DESC;  -- multiplas colunas

Valores unicos

SELECT DISTINCT status FROM tarefas;      -- 3 linhas, sem repeticao
SELECT DISTINCT id_usuario FROM tarefas;  -- quem tem tarefa?

Paginacao (essencial para APIs!)

SELECT * FROM tarefas ORDER BY id_tarefa LIMIT 10 OFFSET 20;  -- pagina 3

A2 Voce vai usar LIMIT/OFFSET na paginacao da sua API REST.

HAVING — filtrando os grupos

WHERE filtra linhas antes de agrupar. HAVING filtra grupos depois de agregar.

-- Usuarios com MAIS de 2 tarefas
SELECT id_usuario, COUNT(*) AS total
FROM tarefas
GROUP BY id_usuario
HAVING total > 2;

-- WHERE + HAVING juntos
SELECT id_usuario, COUNT(*) AS pendentes
FROM tarefas
WHERE status = 'pendente'      -- 1o: filtra linhas
GROUP BY id_usuario            -- 2o: agrupa
HAVING pendentes >= 2;         -- 3o: filtra grupos
Erro classico
WHERE COUNT(*) > 2 nao existe! Funcao de agregacao so no HAVING.

Ordem logica do SELECT

O SQL executa as clausulas nesta ordem (nao na ordem que voce escreve!):

#ClausulaO que faz
1FROMDefine a(s) tabela(s) de origem
2WHEREFiltra linhas individuais
3GROUP BYAgrupa as linhas restantes
4HAVINGFiltra os grupos formados
5SELECTEscolhe as colunas do resultado
6ORDER BYOrdena o resultado final
7LIMITLimita a quantidade de linhas
SELECT id_usuario, COUNT(*) AS total
FROM tarefas
WHERE status <> 'concluida'
GROUP BY id_usuario
HAVING total > 1
ORDER BY total DESC
LIMIT 5;

Mapa mental do DML

Escrever dados
INSERT cria | UPDATE altera | DELETE remove. Sempre com WHERE (exceto INSERT) e preferencialmente em transacao.
Ler dados
SELECT com WHERE (linhas), GROUP BY (grupos), HAVING (filtro de grupos), ORDER BY (ordem), LIMIT (quantidade).
Regra de seguranca
Todo UPDATE/DELETE comeca como um SELECT. Teste a condicao, confira o COUNT, so entao execute.
Mnemonica
WHERE = filtra LINHAS | HAVING = filtra GRUPOS | Sem WHERE no UPDATE/DELETE = afeta TUDO
